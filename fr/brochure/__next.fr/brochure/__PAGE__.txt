(function () {

  var pages = [].slice.call(doc.querySelectorAll('.pb-sheet'));
  var total = pages.length;
  var courant = 0;
  var derniere = 0;

  // la page la plus proche du milieu de l'ecran
  function recalc() {
    derniere = Date.now();
    var milieu = window.innerHeight / 2;
    var best = 0, bestD = Infinity;
    for (var i = 0; i < total; i++) {
      var r = pages[i].getBoundingClientRect();
      var d = Math.abs((r.top + r.bottom) / 2 - milieu);
      if (d < bestD) { bestD = d; best = i; }
    }
    if (best !== courant) {
      courant = best;
      out.innerHTML = 'page <strong>' + (best + 1) + '</strong> / ' + total;
    }
  }
  var attente = null;
  function demande() {
    var maintenant = Date.now();
    if (maintenant - derniere > 90) { recalc(); return; }
    if (attente) return;
    attente = setTimeout(function () { attente = null; recalc(); }, 90);
  }
  window.addEventListener('scroll', demande, { passive: true });
  window.addEventListener('resize', demande, { passive: true });
  recalc();
})();
