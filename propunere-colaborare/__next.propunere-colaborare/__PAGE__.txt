(function () {

  var out = document.getElementById('pbCount');
  if (!doc || !out) return;
  var file = [].slice.call(doc.querySelectorAll('.pb-sheet'));
  var total = file.length;
  var curent = 0;
  var ultima = 0;

  // fila cea mai apropiată de mijlocul ecranului
  function recalc() {
    ultima = Date.now();
    var mijloc = window.innerHeight / 2;
    var best = 0, bestD = Infinity;
    for (var i = 0; i < total; i++) {
      var r = file[i].getBoundingClientRect();
      var d = Math.abs((r.top + r.bottom) / 2 - mijloc);
      if (d < bestD) { bestD = d; best = i; }
    }
    if (best !== curent) {
      curent = best;
      out.innerHTML = 'pagina <strong>' + (best + 1) + '</strong> / ' + total;
    }
  }
  var pendinte = null;
  function cere() {
    var acum = Date.now();
    if (acum - ultima > 90) { recalc(); return; }
    if (pendinte) return;
    pendinte = setTimeout(function () { pendinte = null; recalc(); }, 90);
  }
  window.addEventListener('scroll', cere, { passive: true });
  window.addEventListener('resize', cere, { passive: true });
  recalc();
})();
